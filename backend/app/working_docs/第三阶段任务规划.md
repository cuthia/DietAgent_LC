### 任务 3.4：LCEL 链式调用编排（核心任务 | 预估 4h）

**目标**：使用 LangChain Expression Language 搭建 Agent 处理流程

**文件路径**：`backend/app/agent/chain.py`

#### 核心设计

```python
# 任务3.4: chain.py 完整结构

class DietAgentChain:
    """
    膳食搭配Agent - LCEL链式调用
    
    执行流程：
    用户查询 → 信息分析 → 知识检索 → 约束校验 → 食谱生成 → 结果输出
    
    特点：
    1. 纯函数式链式调用，无状态副作用
    2. 每个环节可独立替换和测试
    3. 支持同步和异步两种模式
    """
    
    def __init__(self, llm_client, retriever, tools):
        """
        初始化Agent链
        
        参数：
        - llm_client: LLM客户端实例
        - retriever: RAG检索器实例
        - tools: 工具集（user_tool, validate_tool等）
        """
        self.llm = llm_client.get_llm()
        self.retriever = retriever
        self.tools = tools
        self._build_pipeline()
    
    def _build_pipeline(self):
        """
        构建完整处理流水线
        
        使用 LCEL 的 Runnable 组合能力：
        - RunnablePassthrough: 透传输入
        - RunnableLambda: 包装自定义函数
        - 管道操作符 |: 串联各个步骤
        """
        
        # Step 1: 知识检索子链
        # 输入: user_query -> 输出: retrieved_docs
        self.retrieval_chain = (
            RunnableLambda(self._prepare_query)      # 处理查询
            | self.retriever                          # 向量检索
            | RunnableLambda(self._format_docs)      # 格式化结果
        )
        
        # Step 2: 约束校验子链
        # 输入: (user_profile, diet_candidate) -> 输出: validation_result
        self.validation_chain = RunnableLambda(
            lambda input: self._validate_constraints(**input)
        )
        
        # Step 3: 食谱生成子链
        # 输入: {query, profile, context, constraints} -> 输出: diet_plan
        self.generation_chain = (
            {"query": RunnablePassthrough(),
             "profile": RunnablePassthrough(),
             "knowledge": self.retrieval_chain}
            | DIET_GENERATE_PROMPT
            | self.llm
            | StrOutputParser()
            | RunnableLambda(self._parse_diet_plan)  # JSON解析
        )
    
    async def process(self, user_query: str, user_profile: dict) -> dict:
        """
        执行完整的膳食搭配流程
        
        这是Agent的主入口方法
        """
        try:
            # 1. 检索相关知识
            knowledge_docs = await self.retrieval_chain.ainvoke(user_query)
            
            # 2. 检查用户信息是否充足
            missing_info = self._check_missing_info(user_profile)
            if missing_info:
                return {
                    "status": "need_info",
                    "message": f"为了更好地为您定制方案，还需要了解：{', '.join(missing_info)}",
                    "missing_fields": missing_info
                }
            
            # 3. 校验约束条件
            constraints = self._build_constraints(user_profile)
            
            # 4. 生成膳食方案
            diet_plan = await self.generation_chain.ainvoke({
                "query": user_query,
                "profile": str(user_profile),
                "knowledge": knowledge_docs,
                "constraints": constraints
            })
            
            # 5. 校验生成结果
            validation = self.validation_chain.invoke({
                "user_profile": user_profile,
                "diet_plan": diet_plan
            })
            
            # 6. 如果校验不通过，触发修正（最多2次）
            retry_count = 0
            while not validation["passed"] and retry_count < 2:
                diet_plan = await self._revise_diet_plan(
                    diet_plan, validation, user_profile, knowledge_docs
                )
                validation = self.validation_chain.invoke({
                    "user_profile": user_profile,
                    "diet_plan": diet_plan
                })
                retry_count += 1
            
            return {
                "status": "success",
                "diet_plan": diet_plan,
                "knowledge_used": knowledge_docs,
                "validation": validation
            }
            
        except Exception as e:
            logger.error(f"Agent处理异常: {e}")
            return {
                "status": "error",
                "message": "生成膳食方案时出现问题，请稍后重试"
            }
    
    def _check_missing_info(self, user_profile: dict) -> list:
        """检查用户档案缺失的关键字段"""
        required_fields = ["age", "gender", "height", "weight"]
        missing = [f for f in required_fields if not user_profile.get(f)]
        return missing
    
    def _build_constraints(self, user_profile: dict) -> dict:
        """构建约束条件"""
        return {
            "taboo": user_profile.get("food_taboo", []),
            "chronic_disease": user_profile.get("chronic_disease", []),
            "region": user_profile.get("region", ""),
            "diet_goal": user_profile.get("diet_goal", "general")
        }
    
    # ... 其他辅助方法
```

# 任务3.5: memory.py

import redis
import json
from datetime import datetime, timedelta

class ConversationMemory:
    """
    对话记忆管理
    
    分层设计：
    - 短期记忆：当前会话历史（Redis List，TTL 24h）
    - 长期记忆：用户偏好存储（Redis Hash，TTL 30天）
    - 摘要压缩：长对话自动摘要（预留接口）
    """
    
    def __init__(self, redis_client=None):
        """
        初始化记忆管理器
        
        如果redis_client为None，则使用内存降级模式
        """
        self.redis = redis_client
        self.message_ttl = timedelta(hours=24)
        self.preference_ttl = timedelta(days=30)
    
    # ========== 会话消息管理 ==========
    
    def add_message(self, session_id: str, role: str, content: str):
        """添加消息到会话历史"""
        if self.redis:
            key = f"session:{session_id}:messages"
            message = json.dumps({
                "role": role,
                "content": content,
                "time": datetime.now().isoformat()
            })
            self.redis.lpush(key, message)
            self.redis.expire(key, self.message_ttl)
        else:
            # 内存降级模式
            if not hasattr(self, '_memory'):
                self._memory = {}
            if session_id not in self._memory:
                self._memory[session_id] = []
            self._memory[session_id].append({
                "role": role, "content": content,
                "time": datetime.now().isoformat()
            })
    
    def get_history(self, session_id: str, max_turns: int = 10) -> list:
        """获取会话历史（最近N轮）"""
        if self.redis:
            key = f"session:{session_id}:messages"
            raw = self.redis.lrange(key, 0, max_turns * 2)
            return [json.loads(msg) for msg in reversed(raw)]
        else:
            history = self._memory.get(session_id, [])
            return history[-max_turns * 2:]
    
    def clear_session(self, session_id: str):
        """清除会话历史"""
        if self.redis:
            self.redis.delete(f"session:{session_id}:messages")
        elif hasattr(self, '_memory') and session_id in self._memory:
            del self._memory[session_id]
    
    # ========== 用户偏好管理 ==========
    
    def get_user_preferences(self, user_id: str) -> dict:
        """获取用户长期偏好"""
        if self.redis:
            data = self.redis.hgetall(f"user:{user_id}:preferences")
            return {k.decode(): json.loads(v) for k, v in data.items()}
        return {}
    
    def update_user_preferences(self, user_id: str, preferences: dict):
        """更新用户偏好"""
        if self.redis:
            key = f"user:{user_id}:preferences"
            for k, v in preferences.items():
                self.redis.hset(key, k, json.dumps(v))
            self.redis.expire(key, self.preference_ttl)
    
    # ========== 摘要压缩（预留） ==========
    
    async def compress_history(self, session_id: str, llm_client) -> str:
        """
        压缩长对话为摘要（预留接口）
        
        当对话超过N轮时，自动生成摘要以节省Token
        """

# 任务3.6.1: agent_service.py

class AgentService:
    """
    Agent 业务服务类
    
    职责：
    1. 编排 Agent 执行流程
    2. 管理会话生命周期
    3. 对接 API 层和 Agent 层
    """
    
    def __init__(self):
        # 初始化各组件
        settings = get_settings()
        self.llm_client = LLMClient(settings.llm)
        self.memory = ConversationMemory(self._get_redis_client())
        self.chain = self._build_agent_chain()
    
    def _get_redis_client(self):
        """获取Redis客户端（可选）"""
        try:
            import redis
            client = redis.Redis(
                host=settings.redis.host,
                port=settings.redis.port,
                db=settings.redis.db
            )
            client.ping()
            return client
        except:
            logger.warning("Redis连接失败，使用内存模式")
            return None
    
    def _build_agent_chain(self) -> DietAgentChain:
        """构建Agent链"""
        retriever = self._get_retriever()
        tools = {
            "user_tool": user_tool,
            "rag_tool": rag_tool,
            "validate_tool": validate_tool,
            "region_tool": region_tool
        }
        return DietAgentChain(self.llm_client, retriever, tools)
    
    async def chat(self, user_id: str, message: str, session_id: str = None) -> dict:
        """
        发送消息给Agent
        
        参数：
        - user_id: 用户ID
        - message: 用户消息
        - session_id: 会话ID（可选，自动生成）
        
        返回：{
            "response": "回复文本",
            "diet_plan": {...},  # 膳食方案
            "session_id": "...",
            "needs_info": False
        }
        """
        # 1. 获取/创建会话
        if not session_id:
            session_id = self._create_session(user_id)
        
        # 2. 记录用户消息
        self.memory.add_message(session_id, "user", message)
        
        # 3. 获取用户档案
        user_profile = await get_user_info(user_id)
        
        # 4. 调用Agent处理
        result = await self.chain.process(message, user_profile)
        
        # 5. 记录AI回复
        self.memory.add_message(session_id, "assistant", result.get("response", ""))
        
        # 6. 返回结果
        return {
            "response": result.get("response", ""),
            "diet_plan": result.get("diet_plan"),

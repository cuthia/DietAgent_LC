**文件**：`frontend/services/api_client.py`

```python
# 任务4.1.2: 统一API客户端

import requests
import json
import logging
from typing import Optional, Dict, Any, Generator
from config import BACKEND_URL, API_PATHS

logger = logging.getLogger(__name__)


class APIClient:
    """
    后端 API 统一客户端
    
    封装所有后端接口调用，处理认证、异常、流式响应
    """
    
    def __init__(self):
        self.base_url = BACKEND_URL
        self.token: Optional[str] = None
        self.user_id: Optional[int] = None
        self.session_id: Optional[str] = None
    
    def _get_headers(self) -> Dict[str, str]:
        """获取请求头（含JWT认证）"""
        headers = {"Content-Type": "application/json"}
        if self.token:
            headers["Authorization"] = f"Bearer {self.token}"
        return headers
    
    def login(self, username: str, password: str) -> Optional[Dict]:
        """用户登录"""
        try:
            resp = requests.post(
                f"{self.base_url}{API_PATHS['login']}",
                json={"username": username, "password": password}
            )
            data = resp.json()
            if data.get("code") == 200:
                self.token = data["data"]["token"]
                self.user_id = data["data"]["user_id"]
                return data["data"]
            return None
        except Exception as e:
            logger.error(f"登录失败: {e}")
            return None
    
    def register(self, username: str, password: str, **kwargs) -> Optional[Dict]:
        """用户注册"""
        ...
    
    def get_profile(self, user_id: int) -> Optional[Dict]:
        """获取用户档案"""
        ...
    
    def update_profile(self, user_id: int, updates: Dict) -> Optional[Dict]:
        """更新用户档案"""
        ...
    
    def chat(self, message: str, session_id: str = None) -> Optional[Dict]:
        """同步对话"""
        ...
    
    def chat_stream(self, message: str, session_id: str = None) -> Generator[Dict, None, None]:
        """
        流式对话（SSE）
        
        Yields:
            每个事件的字典，如：
            {"stage": "collect_info", "status": "start", "message": "..."}
            {"stage": "output", "status": "complete", "data": {...}}
        """
        try:
            url = f"{self.base_url}{API_PATHS['chat_stream']}"
            payload = {"user_id": self.user_id, "message": message}
            if session_id:
                payload["session_id"] = session_id
            
            resp = requests.post(
                url, json=payload, headers=self._get_headers(),
                stream=True
            )
            resp.raise_for_status()
            
            # 解析 SSE 事件
            for line in resp.iter_lines(decode_unicode=True):
                if line and line.startswith("data: "):
                    data_str = line[6:]
                    if data_str == "[DONE]":
                        break
                    try:
                        event = json.loads(data_str)
                        yield event
                    except json.JSONDecodeError:
                        continue
        except Exception as e:
            logger.error(f"流式对话失败: {e}")
            yield {"stage": "error", "message": str(e)}
    
    def get_chat_history(self, user_id: int, max_messages: int = 20) -> list:
        """获取对话历史"""
        ...
    
    def get_diet_history(self, user_id: int, limit: int = 10) -> list:
        """获取膳食方案历史"""
        ...
    
    def clear_history(self, user_id: int, session_id: str = "default") -> bool:
        """清空对话历史"""
        ...
    
    def validate_plan(self, user_id: int, diet_plan: Dict) -> Optional[Dict]:
        """校验膳食方案"""
        ...
    
    # ========== 知识库接口 ==========
    
    def upload_knowledge(self, file) -> Optional[Dict]:
        """上传知识库文档"""
        ...
    
    def search_knowledge(self, query: str, top_k: int = 5) -> list:
        """检索知识库"""
        ...
    
    def delete_knowledge(self, doc_id: str) -> bool:
        """删除文档"""
        ...
    
    def get_knowledge_stats(self) -> Optional[Dict]:
        """获取知识库统计"""
        ...


# 全局 API 客户端实例
_api_client: Optional[APIClient] = None

def get_api_client() -> APIClient:
    """获取全局 API 客户端（单例）"""
    global _api_client
    if _api_client is None:
        _api_client = APIClient()
    return _api_client
```

```python
# 任务4.1.3: 工具函数

from datetime import datetime
from typing import Dict, Any


def format_datetime(ts: float) -> str:
    """时间戳转可读时间"""
    return datetime.fromtimestamp(ts).strftime("%Y-%m-%d %H:%M")


def format_calories(cal: float) -> str:
    """格式化热量显示"""
    if cal >= 1000:
        return f"{cal/1000:.1f}kcal"
    return f"{cal:.0f}kcal"


def get_meal_emoji(meal_type: str) -> str:
    """获取餐次对应的 emoji"""
    emoji_map = {
        "breakfast": "🌅",
        "lunch": "☀️",
        "dinner": "🌙",
        "snack": "🍎",
    }
    return emoji_map.get(meal_type, "🍽️")


def get_meal_label(meal_type: str) -> str:
    """获取餐次中文标签"""
    label_map = {
        "breakfast": "早餐",
        "lunch": "午餐",
        "dinner": "晚餐",
        "snack": "加餐",
    }
    return label_map.get(meal_type, meal_type)


def safe_get(d: Dict, *keys, default=None):
    """安全获取嵌套字典值"""
    current = d
    for key in keys:
        if isinstance(current, dict):
            current = current.get(key)
        else:
            return default
        if current is None:
            return default
    return current
```

#### 4.3.2 核心实现

```python
# 任务4.3: pages/1_💬_膳食对话.py

import streamlit as st
import time
from services.api_client import get_api_client
from components.chat_display import render_chat_message
from components.diet_card import render_diet_plan_card

st.set_page_config(page_title="膳食对话", page_icon="💬", layout="wide")

# ========== 状态初始化 ==========
if "messages" not in st.session_state:
    st.session_state.messages = []
if "processing" not in st.session_state:
    st.session_state.processing = False
if "current_plan" not in st.session_state:
    st.session_state.current_plan = None

api = st.session_state.api_client

# 未登录检查
if not st.session_state.user_id:
    st.warning("请先登录")
    st.stop()

st.title("💬 AI 膳食顾问")

# ========== 聊天消息展示区 ==========
chat_container = st.container(height=500)

with chat_container:
    # 欢迎消息
    if not st.session_state.messages:
        with st.chat_message("assistant", avatar="🥗"):
            st.write("""
            你好！我是 **AI 膳食顾问** 🥗
            
            我可以根据您的健康状况（年龄、体重、慢病、忌口等）为您制定个性化膳食方案。
            
            **试试这样问我：**
            - "给我设计一份减脂餐"
            - "糖尿病患者早餐吃什么好？"
            - "我是北方人，痛风该怎么吃？"
            """)
    
    # 历史消息
    for msg in st.session_state.messages:
        render_chat_message(msg)

# ========== 用户输入区 ==========
if prompt := st.chat_input("描述您的饮食需求...", disabled=st.session_state.processing):
    # 添加用户消息
    st.session_state.messages.append({
        "role": "user",
        "content": prompt,
        "timestamp": time.time()
    })
    
    # 重新渲染显示
    with chat_container:
        render_chat_message({"role": "user", "content": prompt})
    
    # 调用 Agent（流式）
    st.session_state.processing = True
    
    with chat_container:
        with st.chat_message("assistant", avatar="🥗"):
            # 进度显示区
            progress_placeholder = st.empty()
            plan_placeholder = st.empty()
            text_placeholder = st.empty()
            
            progress_messages = []
            
            for event in api.chat_stream(
                message=prompt,
                session_id=st.session_state.session_id
            ):
                if event.get("stage") == "error":
                    text_placeholder.error(f"❌ 处理失败：{event.get('message', '未知错误')}")
                    break
                
                if event.get("status") == "start":
                    progress_messages.append(f"⏳ {event.get('message', '')}")
                elif event.get("status") == "complete":
                    progress_messages.append(f"✅ {event.get('message', '')}")
                
                # 更新进度显示
                with progress_placeholder.container():
                    for msg in progress_messages:
                        st.write(msg)
                
                # 处理最终输出
                if event.get("stage") == "output" and event.get("data"):
                    data = event["data"]
                    
                    # 保存 session_id
                    if data.get("session_id"):
                        st.session_state.session_id = data["session_id"]
                    
                    # 显示文本回复
                    if data.get("message"):
                        text_placeholder.write(data["message"])
                    
                    # 渲染膳食方案
                    if data.get("diet_plan"):
                        st.session_state.current_plan = data["diet_plan"]
                        with plan_placeholder.container():
```

```python
def calculate_bmi(height: float, weight: float) -> float:
    """计算 BMI"""
    if height <= 0:
        return 0
    height_m = height / 100
    return round(weight / (height_m * height_m), 1)


def get_bmi_category(bmi: float) -> str:
    """获取 BMI 分类"""
    if bmi < 18.5:
        return "偏瘦"
    elif bmi < 24:
        return "正常"
    elif bmi < 28:
        return "超重"
    else:
        return "肥胖"
```

```python
def render_nutrition_radar(nutrition: Dict[str, Any]):
    """

        fill="toself",
        name="营养均衡",
        line_color="#2ecc71",
    ))

        height=350,
        margin=dict(l=20, r=20, t=20, b=20),
```
